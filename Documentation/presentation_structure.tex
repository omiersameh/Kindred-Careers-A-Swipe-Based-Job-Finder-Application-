\section{1. Introduction \& System Architecture}
\begin{frame}{1. Introduction \& System Architecture (Member 1 - Architect)}
    \textbf{Speaker:} Member 1 (Project Manager \& System Architect)
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Problem Statement:} Traditional job hunting is tedious. Resumes aren't optimized.
        \item \textbf{Our Solution:} Kindred Careers - A swipe-based, Tinder-like job app.
        \item \textbf{High-Level Architecture:}
        \begin{itemize}
            \item Frontend: Flutter Mobile App
            \item Backend: FastAPI Python Server
            \item AI/ML: Groq LLMs, ChromaDB Vector Store
            \item Data: Automated Camoufox Web Scraper
        \end{itemize}
        \item \textbf{Deployment:} Fully Dockerized offline-capable pipeline.
    \end{itemize}
\end{frame}

\section{3. Backend API \& State Management}
\begin{frame}{3. Backend API \& State Management (Member 3 - Backend Engineer)}
    \textbf{Speaker:} Member 3 (Backend Engineer)
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Tech Stack:} FastAPI, Python, SQLite, Pydantic, Uvicorn.
        \item \textbf{Key Features:}
        \begin{itemize}
            \item RESTful endpoints for Job Feed, Authentication, and Recommendations.
            \item \textbf{Swipe Store:} Persistent tracking of likes/dislikes/skips via SQLite.
            \item \textbf{Rate Limiting:} Memory-gated rolling window quotas.
        \end{itemize}
        \item \textbf{Challenges:} Ensuring stateless horizontal scalability while managing persistent SQLite swipe tracking.
    \end{itemize}
\end{frame}

% =========================================================
\section{4. Data Ingestion \& Stealth Web Scraping}
\begin{frame}{4. Data Ingestion \& Scraping (Member 4 - Data Engineer)}
    \textbf{Speaker:} Member 4 (Data \& Scraping Engineer)
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Tech Stack:} Python AsyncIO, Camoufox (Playwright stealth browser).
        \item \textbf{Key Features:}
        \begin{itemize}
            \item Stream ingestion pipeline running as a background worker.
            \item Defeats anti-bot measures using Camoufox and DOM parsing.
            \item Role-first query routing (dynamically maps user profiles to Wuzzuf queries).
        \end{itemize}
        \item \textbf{Challenges:} Handling Cloudflare timeouts, concurrent page bounding via semaphores.
    \end{itemize}
\end{frame}

% =========================================================
\section{5. RAG Pipeline \& Vector Database}
\begin{frame}{5. RAG Pipeline \& Vector Database (Member 5 - ML Engineer)}
    \textbf{Speaker:} Member 5 (ML \& RAG Engineer)
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Tech Stack:} ChromaDB, HuggingFace Sentence-Transformers.
        \item \textbf{Key Features:}
        \begin{itemize}
            \item Embedding job descriptions into high-dimensional vectors (all-MiniLM-L6-v2).
            \item Cosine similarity search against the user's embedded profile.
            \item Exclusion filtering (removing already-swiped jobs before vector math).
        \end{itemize}
        \item \textbf{Challenges:} Optimizing latency for real-time recommendations and managing local persistent vector storage.
    \end{itemize}
\end{frame}

% =========================================================
\section{6. Generative AI \& Automated CV Builder}
\begin{frame}{6. Generative AI \& Automated CV Builder (Member 6 - GenAI Specialist)}
    \textbf{Speaker:} Member 6 (Generative AI Specialist)
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Tech Stack:} Groq API, OpenRouter, LLaMA 3.3/70B, ReportLab (PDFs).
        \item \textbf{Key Features:}
        \begin{itemize}
            \item Structured extraction (HTML to strictly validated JSON).
            \item \textbf{Tailored CVs:} AI rewrites achievements to match target job descriptions.
            \item PDF generation via ReportLab for ATS-compliant resumes.
        \end{itemize}
        \item \textbf{Challenges:} Prompt engineering for strict JSON adherence, hallucination mitigation.
    \end{itemize}
\end{frame}
